def poly_interp(N):
  f = lambda x: 1 / (1 + (10 * x) ** 2)
  a, b = -1, 1
  xeval = np.linspace(a, b, 1001)

  plt.figure()
  plt.plot(xeval, f(xeval), 'k--', label='Exact f(x)')

  for n_val in [N - 1, N, N + 1]:
    if n_val < 2:
      continue
    h = (b - a) / (n_val - 1)
    x = np.array([-1 + (i - 1) * h for i in range(1, n_val + 1)])
    y = f(x)

    V = np.vander(x, n_val, increasing=True)
    c = np.linalg.solve(V, y)
    peval = np.vander(xeval, n_val, increasing=True) @ c

    print(f'N = {n_val:2d} | Max p(x) = {np.max(np.abs(peval)):.2f}')

    plt.plot(xeval, peval, label=f'Polynomial (N = {n_val})')

    if n_val == N:
        plt.plot(x, y, 'o', label=f'Data points (N = {n_val})')

  plt.legend()
  plt.show()

def barycentric(N):
  f = lambda x: 1 / (1 + (10 * x) ** 2)
  a, b = -1, 1

  h = (b - a) / (N - 1)
  x_nodes = np.array([-1 + (i - 1) * h for i in range(1, N + 1)])
  y_nodes = f(x_nodes)

  w = np.zeros(N)
  for j in range(N):
    diffs = x_nodes[j] - x_nodes
    diffs[j] = 1.0
    w[j] = 1.0 / np.prod(diffs)

  xeval = np.linspace(a, b, 1001)
  peval = np.zeros_like(xeval)

  for idx, xe in enumerate(xeval):
    match = np.isclose(xe, x_nodes)
    if np.any(match):
      peval[idx] = y_nodes[match][0]
    else:
      terms = w / (xe - x_nodes)
      peval[idx] = np.sum(terms * y_nodes) / np.sum(terms)

  # Plotting
  plt.figure()
  plt.plot(xeval, f(xeval), 'k--', label='Exact f(x)')
  plt.plot(xeval, peval, label=f'Barycentric (N = {N})')
  plt.plot(x_nodes, y_nodes, 'o', label='Data points')
  plt.legend()
  plt.show()

  a, b = -1, 1

  j = np.arange(1, N+1)
  x_nodes = np.cos(((2*j - 1) * np.pi)/(2 * N))
